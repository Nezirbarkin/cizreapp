-- DİKKAT: Tablo seviyesi GRANT, sütun seviyesi GRANT'leri YUTAR.
-- 20260907110002 tablo SELECT'ini geri verdiği için 20260907110001'in
-- sütun GRANT'leri artık ayrı kayıt olarak durmuyor. Bu canlıda
-- ROLLBACK'li bir dry-run ile doğrulandı (2026-09-08): yalnız REVOKE
-- çalıştırılsaydı `authenticated` profiles üzerinde SIFIR sütun görecek
-- ve uygulama tamamen kırılacaktı.
--
-- Bu yüzden REVOKE ve GRANT AYNI transaction içinde, sırayla yapılır.

BEGIN;

REVOKE SELECT ON TABLE public.profiles FROM authenticated;

GRANT SELECT (
  -- kimlik / görünen profil
  id, username, full_name, avatar_url, banner_url, cover_url,
  bio, website, location, gender,
  -- rol & durum
  role, status, is_verified, delivered_count,
  -- varlık / gizlilik tercihleri
  is_online, is_online_enabled, is_ghost_mode, show_last_seen, last_seen,
  profile_is_public, messages_enabled, allow_messages_from_non_followers,
  -- zaman damgaları (public_profiles_safe / public_profiles_chat kullanıyor)
  created_at, updated_at,
  -- OAuth kayıt akışı (20260916000001_oauth_signup_needs_username.sql)
  needs_username
) ON public.profiles TO authenticated;

COMMIT;

NOTIFY pgrst, 'reload schema';
