-- ****************************************************************************
-- UYGULAMA ZAMANI: Order/OrderItem/ShopReview modellerinin tarihleri
-- .toLocal() ile ayrıştırdığı istemci sürümü (2026-10-17 değişikliği)
-- mağazalarda yayınlandıktan HEMEN SONRA. Önce uygulanırsa eski sürüm
-- (UTC alanlarını okuyan ekranlar) eski siparişleri 3 saat geri gösterir.
-- Uygulama: dosyayı supabase/migrations/2026MMDD00000N_server_created_at_and_tz_backfill.sql
-- olarak yeniden adlandırıp `supabase db query --linked --file …` ile çalıştır.
-- Test: supabase/tests/manual/server_created_at_test.sql (14 sipariş kayar).
-- ****************************************************************************
-- ============================================================================
-- İstemci zaman damgası: created_at sunucuda (2026-10-17)
-- ----------------------------------------------------------------------------
-- İstemci 'created_at': DateTime.now().toIso8601String() yazıyordu. Yerel
-- (UTC+3) DateTime offset'siz serileşir; PostgREST oturumu UTC olduğundan
-- değer UTC sanılır ve kayıt 3 saat İLERİ kayar (posts için 2026-09-22'de
-- düzeltilen kalıp: 20260922000004). Canlıda doğrulandı: eski (doğrudan
-- insert) checkout'un 14 siparişi ve 4 mağaza yorumu, aynı anda sunucu
-- saatiyle yazılan bildirimlerinden tam +3,00 saat ileride.
--
-- 1) Sahadaki eski sürümler de düzelsin diye istemci (authenticated/anon)
--    INSERT'inde created_at (ve varsa updated_at) her zaman now() olur.
--    DEFINER RPC/tetikleyici ve service_role yazımları etkilenmez.
-- 2) Yalnız bildirimle doğrulanan satırlar 3 saat geri alınır. Orders
--    güncellenirken updated_at tetikleyicileri kapatılır: updated_at
--    değişirse eski teslim edilmiş siparişler get_pending_reviews'ta
--    (30 gün penceresi updated_at'e bakıyor) yeniden "değerlendir" diye
--    açılırdı. Mağaza yorumunda puan tetikleyicileri kapatılır (puan değişmiyor).
-- ============================================================================

BEGIN;

CREATE OR REPLACE FUNCTION private.force_server_created_at()
RETURNS trigger
LANGUAGE plpgsql
SECURITY INVOKER
SET search_path TO ''
AS $function$
DECLARE
  v_patch jsonb;
BEGIN
  IF current_user NOT IN ('authenticated', 'anon') THEN
    RETURN NEW;
  END IF;

  v_patch := jsonb_build_object('created_at', now());
  IF to_jsonb(NEW) ? 'updated_at' THEN
    v_patch := v_patch || jsonb_build_object('updated_at', now());
  END IF;

  NEW := jsonb_populate_record(NEW, v_patch);
  RETURN NEW;
END;
$function$;

REVOKE ALL ON FUNCTION private.force_server_created_at() FROM PUBLIC, anon, authenticated;

DO $$
DECLARE
  v_table text;
BEGIN
  FOREACH v_table IN ARRAY ARRAY[
    'orders', 'order_items', 'shop_reviews',
    'courier_requests', 'follow_requests', 'notifications'
  ] LOOP
    EXECUTE format('DROP TRIGGER IF EXISTS trg_%s_server_created_at ON public.%I', v_table, v_table);
    EXECUTE format(
      'CREATE TRIGGER trg_%s_server_created_at BEFORE INSERT ON public.%I '
      'FOR EACH ROW EXECUTE FUNCTION private.force_server_created_at()',
      v_table, v_table
    );
  END LOOP;
END;
$$;

-- ---------------------------------------------------------------------------
-- Geri dönük düzeltme (yalnız doğrulanan satırlar)
-- ---------------------------------------------------------------------------
CREATE TEMP TABLE _tz_orders ON COMMIT DROP AS
SELECT o.id
FROM public.orders o
JOIN LATERAL (
  SELECT min(n.created_at) AS t
  FROM public.notifications n
  WHERE n.entity_id = o.id::text
    AND n.type IN ('admin_new_order', 'new_order')
) n ON true
WHERE o.checkout_session_id IS NULL
  AND n.t IS NOT NULL
  AND o.created_at - n.t BETWEEN interval '2 hours 55 minutes' AND interval '3 hours 5 minutes';

ALTER TABLE public.orders DISABLE TRIGGER update_orders_updated_at;
ALTER TABLE public.orders DISABLE TRIGGER orders_updated_at_trigger;

UPDATE public.orders o
   SET created_at = o.created_at - interval '3 hours'
  FROM _tz_orders t
 WHERE o.id = t.id;

ALTER TABLE public.orders ENABLE TRIGGER update_orders_updated_at;
ALTER TABLE public.orders ENABLE TRIGGER orders_updated_at_trigger;

CREATE TEMP TABLE _tz_reviews ON COMMIT DROP AS
SELECT r.id
FROM public.shop_reviews r
JOIN LATERAL (
  SELECT min(n.created_at) AS t
  FROM public.notifications n
  WHERE n.type = 'shop_review'
    AND n.entity_id IN (r.id::text, r.shop_id::text)
    AND n.created_at BETWEEN r.created_at - interval '4 hours' AND r.created_at + interval '4 hours'
) n ON true
WHERE n.t IS NOT NULL
  AND r.created_at - n.t BETWEEN interval '2 hours 55 minutes' AND interval '3 hours 5 minutes';

ALTER TABLE public.shop_reviews DISABLE TRIGGER shop_reviews_updated_at;
ALTER TABLE public.shop_reviews DISABLE TRIGGER shop_reviews_insert_rating;
ALTER TABLE public.shop_reviews DISABLE TRIGGER update_shop_rating_trigger;
ALTER TABLE public.shop_reviews DISABLE TRIGGER notify_user_seller_reply_trigger;

UPDATE public.shop_reviews r
   SET created_at = r.created_at - interval '3 hours'
  FROM _tz_reviews t
 WHERE r.id = t.id;

ALTER TABLE public.shop_reviews ENABLE TRIGGER shop_reviews_updated_at;
ALTER TABLE public.shop_reviews ENABLE TRIGGER shop_reviews_insert_rating;
ALTER TABLE public.shop_reviews ENABLE TRIGGER update_shop_rating_trigger;
ALTER TABLE public.shop_reviews ENABLE TRIGGER notify_user_seller_reply_trigger;

COMMIT;

NOTIFY pgrst, 'reload schema';
